async function changeMode(child,fixture) {
  if(!child.connected)throw new Error('QA fixture process unavailable');
  await new Promise((resolve,reject)=>{child.once('message',resolve);child.send({fixture},error=>{if(error)reject(error)});});
}
async function tokenFromRealCookies(context) {
  const cookies=(await context.cookies()).filter(cookie=>/sb-.*-auth-token(?:\.\d+)?$/.test(cookie.name)).sort((a,b)=>a.name.localeCompare(b.name));
  const encoded=cookies.map(cookie=>cookie.value).join('');
  return encoded ? JSON.parse(encoded.startsWith('base64-')?Buffer.from(encoded.slice(7),'base64url').toString('utf8'):encoded).access_token : null;
}
